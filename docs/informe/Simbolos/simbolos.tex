%% LISTA DE SÍMBOLOS

\nomenclature{$n$}{Número de trabajos de la instancia}
\nomenclature{$m$}{Número de máquinas de la instancia}
\nomenclature{$J_i$}{Trabajo \(i\), con \(i=1,\ldots,n\)}
\nomenclature{$O_{ij}$}{Operación \(j\)-ésima del trabajo \(i\)}
\nomenclature{$m_{ij}$}{Máquina requerida por la operación \(O_{ij}\)}
\nomenclature{$p_{ij}$}{Duración de la operación \(O_{ij}\)}
\nomenclature{$s_{ij}$}{Instante de inicio de la operación \(O_{ij}\)}
\nomenclature{$C_i$}{Instante de término del trabajo \(i\)}
\nomenclature{$C_{\max}$}{Makespan o instante de término de todos los trabajos}
\nomenclature{$T$}{Temperatura actual del recocido simulado}
\nomenclature{$T_0$}{Temperatura inicial}
\nomenclature{$T_{\min}$}{Temperatura mínima de término}
\nomenclature{$\alpha$}{Factor de enfriamiento geométrico}
\nomenclature{$\Delta$}{Variación del makespan al pasar de una solución a un vecino}
\nomenclature{$x$}{Secuencia de trabajos que representa una solución}
\nomenclature{$x^{*}$}{Mejor secuencia encontrada durante la ejecución}
